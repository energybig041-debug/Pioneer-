function login() { const name = document.getElementById('loginName').value; const pass = document.getElementById('loginPass').value; // If no account exists, auto-create one if(!localStorage.getItem('pioneerUser')) { if(name && pass) { localStorage.setItem('pioneerUser', name); localStorage.setItem('pioneerPass', pass); alert('Welcome to Pioneer, ' + name + '! ⚡ Account created!'); showApp(); } else { alert('Type your name and password first!'); } return; } // If account exists, check it if(name === localStorage.getItem('pioneerUser')) { showApp(); } else { alert('Wrong details! Your account name is: ' + localStorage.getItem('pioneerUser')); } } function showSignup() { const name = prompt('Full Name:'); const pass = prompt('Password:'); if(name && pass) { localStorage.setItem('pioneerUser', name); localStorage.setItem('pioneerPass', pass); alert('Account created! Welcome ' + name + ' ⚡'); showApp(); } }
